# Alles, was die Skripte in diesem Ordner brauchen.
#
# Installieren mit:
#     pip install -r requirements.txt
#
# Das ist absichtlich nur ein einziges Paket. Statistik, CSV-Verarbeitung
# und das Auslesen der HTML-Seiten laufen ueber Pythons eingebaute
# Standardbibliothek — kein pandas, kein numpy, kein BeautifulSoup.
# Damit gibt es beim Aufsetzen praktisch nichts, was schiefgehen kann.

requests>=2.28

# --- Ab hier alles optional -------------------------------------------
# Die Standardquelle ESPN braucht nichts davon. Beide Pakete helfen nur
# bei --source fbref, wo Cloudflare den Zugriff prueft.

# Optional, aber empfohlen: damit requests brotli-komprimierte Antworten
# auspacken kann. Erst dann bietet der Abruf "Accept-Encoding: gzip,
# deflate, br" an — also genau das, was ein echter Chrome schickt. Fehlt
# das Paket, wird "br" automatisch weggelassen; alles laeuft trotzdem.
brotli>=1.0

# Optional, aber der wirksamste Hebel gegen Cloudflare: curl_cffi bildet
# den TLS-Fingerabdruck eines echten Browsers nach. Cloudflare prueft
# naemlich nicht nur die Kopfzeilen, sondern auch den Handshake — und
# der von Python-requests ist sofort als solcher erkennbar. Ist das
# Paket installiert, wird es fuer FBref automatisch benutzt; fehlt es,
# laeuft alles ueber requests weiter.
curl_cffi>=0.7
